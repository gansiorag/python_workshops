### Пример 2: Численный шифр Цезаря
```python
def caesar_cipher(text, shift):
    result = ""

    for char in text:
        if char.isalpha():
            ascii_offset = 65 if char.isupper() else 97
            result += chr((ord(char) - ascii_offset + shift) % 26 + ascii_offset)
        else:
            result += char

    return result

text = "HELLO"
shift = 3
print(caesar_cipher(text, shift))  # Outputs: KHOOR
```

### Пример 3: Генератор случайных целых чисел.
```python
import random
from typing import Tuple

def generate_key(size: int) -> str:
    letters_and_digits = '0123456789ABCDEFGHIJKLMNOPQRSTUVWXYZ'
    key = ''.join(random.choice(letters_and_digits) for _ in range(size))
    return key

# Генерируем ключ длиной 16 символов.
key_size = 16
print(generate_key(key_size))
```

### Пример 4: Асинхронное чтение файлов с помощью aiofiles.
```python
import asyncio
from aiofiles import open as afopen

async def load_file(filename):
    async with afopen(filename, 'r') as f:
        contents = await f.read()
    return contents

filename = "example.txt"
contents = asyncio.run(load_file(filename))
print(contents)
```

### Пример 5: Использование декораторов для логирования.
```python
import functools
import logging

logger = logging.getLogger(__name__)

def log decorated_function(func):
    @functools.wraps(func)
    def wrapper(*args, **kwargs):
        try:
            result = func(*args, **kwargs)
        except Exception as e:
            logger.exception(f"ОШИБКА: {func.__qualname__}: {e}")
        else:
            logger.info(f"SUCCESS: {func.__qualname__} executed")
            return result
    return wrapper

@log
def some_function():
    pass

# Use the decorated function.
some_function()
```
